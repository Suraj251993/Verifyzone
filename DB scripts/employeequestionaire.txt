-- Table: orgcheck.employeequestionaire

-- DROP TABLE IF EXISTS orgcheck.employeequestionaire;

CREATE TABLE IF NOT EXISTS orgcheck.employeequestionaire
(
    id integer NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    employeeid integer NOT NULL,
    questionid integer NOT NULL,
    answer text COLLATE pg_catalog."default" NOT NULL,
    status integer NOT NULL DEFAULT 1,
    CONSTRAINT employeequestionaire_pkey PRIMARY KEY (id),
    CONSTRAINT empqn_employee FOREIGN KEY (employeeid)
        REFERENCES orgcheck.employee (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
        NOT VALID,
    CONSTRAINT empqn_question FOREIGN KEY (questionid)
        REFERENCES orgcheck.questionaire (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
        NOT VALID
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS orgcheck.employeequestionaire
    OWNER to postgres;
